(function () { var referrerHref = null; if (document.referrer) { try { var referrerUrl = new URL(document.referrer, window.location.origin); if (referrerUrl.origin === window.location.origin) { referrerHref = referrerUrl.href; } } catch (error) { referrerHref = null; } } document.querySelectorAll('[data-hio-back-button]').forEach(function (button) { if (referrerHref) { button.href = referrerHref; } button.addEventListener('click', function (event) { event.preventDefault(); if (referrerHref) { var fallback = setTimeout(function () { window.location.href = button.href; }, 200); window.addEventListener('beforeunload', function () { clearTimeout(fallback); }, { once: true }); window.history.back(); return; } window.location.href = button.href; }); }); })();
